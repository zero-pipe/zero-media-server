set(ZMS_DEMO_SERVER_RUNTIME demo_server_runtime.c)

function(zms_add_demo name)
    add_executable(${name} ${name}.c)
    # ZMS is static on MSVC: executables must link ztk for symbols used inside ZMS (e.g. ztk_info)
    target_link_libraries(${name} PRIVATE zms ztk)
    if(MSVC)
        target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endfunction()

function(zms_add_server_demo name)
    add_executable(${name} ${name}.c ${ZMS_DEMO_SERVER_RUNTIME})
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(${name} PRIVATE zms ztk)
    if(MSVC)
        target_compile_definitions(${name} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endfunction()

zms_add_demo(demo_rtsp_player)

zms_add_demo(hello_server)
zms_add_server_demo(demo_media_server)
zms_add_server_demo(demo_rtsp_server)
zms_add_server_demo(demo_rtmp_server)
